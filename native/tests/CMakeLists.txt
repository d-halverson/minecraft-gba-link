cmake_minimum_required(VERSION 3.16)

set(GBA_TESTS_ROM_DIR ${CMAKE_SOURCE_DIR}/third_party/gba-tests)
set(GBA_TEST_OUT ${CMAKE_BINARY_DIR}/test-output)
file(MAKE_DIRECTORY ${GBA_TEST_OUT})

function(gba_test_exe name)
  add_executable(${name} ${name}.c)
  target_link_libraries(${name} PRIVATE gbacore_obj)
  target_include_directories(${name} PRIVATE ${CMAKE_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/include)
endfunction()

gba_test_exe(test_headless)

# Expected frame hashes of the final screen. jsmolka's tests print "All tests passed" or
# "Failed test N"; mGBA 0.10.5 + HLE BIOS is known to fail a few exotic edge-case tests:
#   arm.gba   -> "Failed test 235"  (CMP with Rd=PC does not flush the pipeline)
#   thumb.gba -> "Failed test 230"  (STM with base register in rlist)
#   sram/flash saves -> "Failed test 006" (unaligned halfword store to SRAM is ignored by mGBA)
#   memory.gba, save/none.gba -> "All tests passed"
function(gba_headless name rom hash)
  add_test(NAME headless_${name}
    COMMAND test_headless ${GBA_TESTS_ROM_DIR}/${rom} ${GBA_TEST_OUT}/${name}.ppm ${hash} ${ARGN})
endfunction()
gba_headless(memory memory/memory.gba 25ce63b225d822c3)
gba_headless(arm arm/arm.gba b40b7ef6b0a6a336)
gba_headless(thumb thumb/thumb.gba e68b1825688d12b3)
gba_headless(none save/none.gba 25ce63b225d822c3)
gba_headless(hello ppu/hello.gba 1b7e4389479b8bbe)
gba_headless(stripes ppu/stripes.gba - nosave 5A84AD)
gba_headless(sram save/sram.gba f4555cb8b0c50bae save)

gba_test_exe(test_rtc)
add_test(NAME rtc COMMAND test_rtc ${GBA_TESTS_ROM_DIR}/ppu/hello.gba)

# ---- link tests: ROMs generated by tests/make_link_rom.py ----
find_package(Python3 COMPONENTS Interpreter)
set(LINK_ROM_DIR ${CMAKE_BINARY_DIR}/test-roms)
add_custom_command(OUTPUT ${LINK_ROM_DIR}/link_multi.gba
  COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_CURRENT_SOURCE_DIR}/make_link_rom.py ${LINK_ROM_DIR}
  DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/make_link_rom.py)
add_custom_target(link_roms ALL DEPENDS ${LINK_ROM_DIR}/link_multi.gba)
gba_test_exe(test_link)
add_dependencies(test_link link_roms)
add_test(NAME link COMMAND test_link ${LINK_ROM_DIR})

gba_test_exe(test_group)
add_dependencies(test_group link_roms)
add_test(NAME group COMMAND test_group ${LINK_ROM_DIR})

gba_test_exe(test_determinism)
add_dependencies(test_determinism link_roms)
add_test(NAME determinism COMMAND test_determinism ${LINK_ROM_DIR})

# ---- shared library: exported symbols + dlopen smoke test ----
add_executable(test_dylib test_dylib.c)
if(UNIX)
  target_link_libraries(test_dylib PRIVATE ${CMAKE_DL_LIBS})
endif()
add_dependencies(test_dylib gbacore)
add_test(NAME dylib COMMAND test_dylib $<TARGET_FILE:gbacore> ${GBA_TESTS_ROM_DIR}/ppu/hello.gba)
if(APPLE)
  add_test(NAME exports
    COMMAND sh -c "test -z \"$(nm -gU $<TARGET_FILE:gbacore> | grep -v ' _gba_')\"")
elseif(UNIX)
  add_test(NAME exports
    COMMAND sh -c "test -z \"$(nm -D --defined-only $<TARGET_FILE:gbacore> | grep -v ' gba_')\"")
endif()
